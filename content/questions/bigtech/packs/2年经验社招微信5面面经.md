### 设定关系（A=1,B=2,C=3, $\cdots$ , Z=26,AA=27,AB=28, $\cdots$ , AAA=XXX, $\cdots$ ），写一个转换函数，根据上面规则把一个字符串转换为数字.

```javascript
const convert = (str) => {
    let res = 0;
    const strArr = str.split('');
    let j;

    const { length } = strArr;
    for(let i = length - 1, j = 1; i >= 0; i--, j *= 26){
    const value = strArr[i].toUpperCase();
    if( value < 'A' || value > 'Z') {
    return 0;
    }
    //value.charCodeAt(0) - 64可以算出A=1,B=2,C=3等对应关系
    res = res + (value.charCodeAt(0) - 64) * j
    }
    return res
```

### 两个有序（从小到大）单链表{value:Number;next:Object}，找出这两条链表中value相等的节点，构造成一条新的链表。

```javascript
const commonNodeList = (l1, l2) => {
    const newNodeList = new ListNode(0)
    // 哪个 l 的 value 比较小就往 next 移
    while (l1 && l2) {
    if (l1.value === l2.value) {
    newNodeList.next = l1.value;
    l1 = l1.next;
    l2 = l2.next;
    } else if (l1.value < l2.value) {
    l1 = l1.next;

    l2 = l2.next;
    }
    }
    return newNodeList
```
